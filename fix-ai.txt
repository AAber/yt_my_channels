import 'dart:convert';
import 'package:http/http.dart' as http;

class GroqModel {
  final String id;
  final String ownedBy;

  GroqModel({required this.id, required this.ownedBy});

  factory GroqModel.fromJson(Map<String, dynamic> json) => GroqModel(
        id: json['id'] as String,
        ownedBy: json['owned_by'] as String? ?? '',
      );
}

class GroqModelResolver {
  final String apiKey;
  final Duration cacheTtl;

  GroqModelResolver({required this.apiKey, this.cacheTtl = const Duration(hours: 6)});

  static List<GroqModel>? _cache;
  static DateTime? _cachedAt;

  /// Ordered list of models you're happy to use, most preferred first.
  /// If the top choice isn't in the active list (deprecated), it falls
  /// through to the next, and finally to any active model at all.
  static const List<String> _preferredOrder = [
    'qwen/qwen3-32b',
    'llama-3.3-70b-versatile',
    'llama-3.1-8b-instant',
  ];

  Future<List<GroqModel>> fetchActiveModels({bool forceRefresh = false}) async {
    if (!forceRefresh &&
        _cache != null &&
        _cachedAt != null &&
        DateTime.now().difference(_cachedAt!) < cacheTtl) {
      return _cache!;
    }

    final response = await http
        .get(
          Uri.parse('https://api.groq.com/openai/v1/models'),
          headers: {'Authorization': 'Bearer $apiKey'},
        )
        .timeout(const Duration(seconds: 10));

    if (response.statusCode != 200) {
      throw Exception('Failed to fetch Groq models: ${response.statusCode}');
    }

    final data = jsonDecode(response.body) as Map<String, dynamic>;
    final models = (data['data'] as List)
        .map((m) => GroqModel.fromJson(m as Map<String, dynamic>))
        .toList();

    _cache = models;
    _cachedAt = DateTime.now();
    return models;
  }

  /// Returns the best available model ID, or a safe fallback if the
  /// network call fails entirely.
  Future<String> resolveModel({String fallback = 'llama-3.1-8b-instant'}) async {
    try {
      final models = await fetchActiveModels();
      final activeIds = models.map((m) => m.id).toSet();

      for (final preferred in _preferredOrder) {
        if (activeIds.contains(preferred)) return preferred;
      }

      // None of our preferred models are live anymore — grab whatever is.
      if (activeIds.isNotEmpty) return activeIds.first;

      return fallback;
    } catch (_) {
      return fallback;
    }
  }
}

class AgentConfig {
  final String groqApiKey;
  final String model;
  final int maxTokens;
  final double temperature;
  final Duration timeout;

  AgentConfig({
    required this.groqApiKey,
    required this.model,
    this.maxTokens = 1000,
    this.temperature = 0.7,
    this.timeout = const Duration(seconds: 30),
  });

  static Future<AgentConfig> create({
    required String groqApiKey,
    int maxTokens = 1000,
    double temperature = 0.7,
    Duration timeout = const Duration(seconds: 30),
  }) async {
    final resolver = GroqModelResolver(apiKey: groqApiKey);
    final model = await resolver.resolveModel();
    return AgentConfig(
      groqApiKey: groqApiKey,
      model: model,
      maxTokens: maxTokens,
      temperature: temperature,
      timeout: timeout,
    );
  }
}
